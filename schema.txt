/* =========================
   SCHEMA AND SEED DATA
   Shared by the whole team. Drops and recreates all tables and sequences,
   then loads sample data. Run this first.
   The DROP statements print errors the first time you run it (nothing
   exists yet to drop). That is expected.
   ========================= */
SET SERVEROUTPUT ON;
DROP TABLE recommendations cascade constraints;
DROP TABLE reviews cascade constraints;
DROP TABLE orders cascade constraints;
DROP TABLE inventory cascade constraints;
DROP TABLE waiters cascade constraints;
DROP TABLE customers cascade constraints;
DROP TABLE menuItems cascade constraints;
DROP TABLE restaurant cascade constraints;
DROP TABLE cuisinetypes cascade constraints;

drop sequence seq_cuisine_id;
drop sequence seq_restaurant_id;
drop sequence seq_waiter_id;
drop sequence seq_menuitem_id;
drop sequence seq_inventory_id;
drop sequence seq_customer_id;
drop sequence seq_order_id;
drop sequence seq_review_id;
drop sequence seq_recommendation_id;

create sequence seq_cuisine_id start with 1 increment by 1;
create sequence seq_restaurant_id start with 1 increment by 1;
create sequence seq_waiter_id start with 1 increment by 1;
create sequence seq_menuitem_id start with 1 increment by 1;
create sequence seq_inventory_id start with 1 increment by 1;
create sequence seq_customer_id start with 1 increment by 1;
create sequence seq_order_id start with 1 increment by 1;
create sequence seq_review_id start with 1 increment by 1;
create sequence seq_recommendation_id start with 1 increment by 1;


create table cuisinetypes (
	cuisine_id int,
	cuisine_name varchar(50),
	primary key (cuisine_id)
);

create table restaurant (
	restaurant_id int,
	name varchar(100),
	streetaddress varchar(100),
	city varchar(50),
	state char(2),
	zip varchar(10),
	cuisine_id int,
	primary key (restaurant_id),
	foreign key (cuisine_id) references cuisinetypes(cuisine_id)
);

create table waiters (
	waiter_id int,
	name varchar(100),
	restaurant_id int,
	primary key (waiter_id),
	foreign key (restaurant_id) references restaurant(restaurant_id)
);

create table menuitems (
	menu_item_id int,
	name varchar(100),
	price number (8,2),
	cuisine_id int,
	primary key (menu_item_id),
	foreign key (cuisine_id) references cuisinetypes(cuisine_id)
);

create table inventory (
	inventory_id int,
	restaurant_id int,
	menu_item_id int,
	item_name varchar(100),
	quantity int,
	primary key (inventory_id),
	foreign key (restaurant_id) references restaurant(restaurant_id),
	foreign key (menu_item_id) references menuitems(menu_item_id)
);


create table customers (
	customer_id int,
	name varchar(100),
	streetaddress varchar(100),
	email varchar(150),
	city varchar(60),
	state char(2),
	zip varchar(10),
	creditcardnumber varchar(20),
	primary key (customer_id)
);

create table orders (
	order_id int,
	orderdate date,
	amount_paid number(8,2),
	tip number (8,2),
	restaurant_id int,
	customer_id int,
	menu_item_id int,
	waiter_id int,
	primary key (order_id),
	foreign key (restaurant_id) references restaurant(restaurant_id),
	foreign key (customer_id) references customers(customer_id),
	foreign key (menu_item_id) references menuitems(menu_item_id),
	foreign key (waiter_id) references waiters(waiter_id)
);

create table reviews (
	review_id int,
	email varchar(150),
	stars int,
	reviewtext varchar(1000),
	restaurant_id int,
	primary key (review_id),
	foreign key (restaurant_id) references restaurant(restaurant_id)
);

create table recommendations (
	recommendations_id int,
	recommendation_date date,
	customer_id int,
	restaurant_id int,
	primary key (recommendations_id),
	foreign key (customer_id) references customers(customer_id),
	foreign key (restaurant_id) references restaurant(restaurant_id)
);
