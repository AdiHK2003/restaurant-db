/* PURPOSE:
   Manages waiter records: look up a waiter's ID by name, add a waiter
   to a restaurant, and list a restaurant's waiters alphabetically.

   DEPENDS ON (created elsewhere in the project):
   - tables: waiters, restaurant
   - sequence: seq_waiter_id
   - function: find_restaurant_id (shared helper)
*/
CREATE OR REPLACE FUNCTION find_waiter_id (
    p_waiter_name IN VARCHAR2
) RETURN INT IS
    v_id INT;
BEGIN
    SELECT waiter_id
    INTO v_id
    FROM waiters
    WHERE LOWER(name) = LOWER(p_waiter_name);

    RETURN v_id;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN NULL;
    WHEN TOO_MANY_ROWS THEN
        DBMS_OUTPUT.PUT_LINE('Error: multiple waiters found');
        RETURN NULL;
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error in find_waiter_id: ' || SQLERRM);
        RETURN NULL;
END;
/
CREATE OR REPLACE PROCEDURE add_waiter (
    p_waiter_name IN VARCHAR2,
    p_restaurant_name IN VARCHAR2
)
IS
    v_restaurant_id INT;
BEGIN
    v_restaurant_id := find_restaurant_id(p_restaurant_name);

    IF v_restaurant_id IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('Error: restaurant not found');
        RETURN;
    END IF;

    INSERT INTO waiters (
        waiter_id,
        name,
        restaurant_id
    )
    VALUES (
        seq_waiter_id.NEXTVAL,
        p_waiter_name,
        v_restaurant_id
    );

    DBMS_OUTPUT.PUT_LINE('Waiter added: ' || p_waiter_name);

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error in add_waiter: ' || SQLERRM);
END;
/
CREATE OR REPLACE PROCEDURE show_waiters_by_restaurant (
    p_restaurant_name IN VARCHAR2
)
IS
    v_restaurant_id INT;

    CURSOR c_waiters IS
        SELECT name
        FROM waiters
        WHERE restaurant_id = v_restaurant_id
        ORDER BY name;

    v_name waiters.name%TYPE;
BEGIN
    v_restaurant_id := find_restaurant_id(p_restaurant_name);

    IF v_restaurant_id IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('Error: restaurant not found');
        RETURN;
    END IF;

    DBMS_OUTPUT.PUT_LINE('--- Waiters at ' || p_restaurant_name || ' ---');

    OPEN c_waiters;
    LOOP
        FETCH c_waiters INTO v_name;
        EXIT WHEN c_waiters%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE('-> ' || v_name);
    END LOOP;
    CLOSE c_waiters;

EXCEPTION
    WHEN OTHERS THEN
        IF c_waiters%ISOPEN THEN
            CLOSE c_waiters;
        END IF;
        DBMS_OUTPUT.PUT_LINE('Error in show_waiters_by_restaurant: ' || SQLERRM);
END;
/
